// Date pop-up. It opens on today; the board shows it after the 12th was tapped, so the chosen day and today can both be seen. // The chosen day is a filled circle. Today is only a coloured number with a dot, so it never looks like a second choice. popVals(t) { const k = this.popKit(t); const st = this.state || {}; const sel = st.popSel != null ? st.popSel : k.dn(2026, 9, 12); const sd = k.dt(sel); const ym = st.popYM != null ? st.popYM : sd.getUTCFullYear() * 12 + sd.getUTCMonth(); const y = Math.floor(ym / 12); const m = ym % 12; const first = k.dn(y, m, 1); const lead = k.dt(first).getUTCDay(); const prev = () => this.setState({ popYM: ym - 1 }); const next = () => this.setState({ popYM: ym + 1 }); // Swiping across the pop-up changes the month, the same as the two arrows. const sw = this.popSwipe(prev, next); // Always six weeks, so Cancel and Confirm never move. Days of the months on either side are shown faintly. const days = []; for (let i = 0; i < 42; i++) { const n = first - lead + i; const d = k.dt(n); const inMonth = d.getUTCMonth() === m; const on = n === sel; const today = n === k.TODAY; days.push({ day: String(d.getUTCDate()), aria: k.WD[d.getUTCDay()] + ', ' + k.ML[d.getUTCMonth()] + ' ' + d.getUTCDate() + (today ? ', today' : ''), on, today, bg: on ? t.accent : 'transparent', dot: on ? t.onAccent : t.accent, fg: on ? t.onAccent : (today ? t.accent : (inMonth ? t.text : k.dim)), fw: on || today ? '800' : '600', pick: this.popTap(() => this.setState({ popSel: n, popYM: d.getUTCFullYear() * 12 + d.getUTCMonth() })) }); } return { head: k.WS[sd.getUTCDay()] + ', ' + k.MS[sd.getUTCMonth()] + ' ' + sd.getUTCDate() + ', ' + sd.getUTCFullYear(), month: k.ML[m] + ' ' + y, week: ['S', 'M', 'T', 'W', 'T', 'F', 'S'].map((label) => ({ label })), days, prev, next, swDown: sw.down, swUp: sw.up }; }
