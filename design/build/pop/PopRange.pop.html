// Custom range pop-up. It opens on the last range used (the last 30 days the first time). // The line under the title always says how long the chosen range is. popVals(t) { const k = this.popKit(t); const st = this.state || {}; const T = k.TODAY; const from = st.popFrom != null ? st.popFrom : T - 29; const to = st.popTo != null ? st.popTo : T; const days = to - from + 1; const short = (n) => { const d = k.dt(n); return k.MS[d.getUTCMonth()] + ' ' + d.getUTCDate(); }; const show = (n) => (n === T ? 'Today, ' + short(n) : k.full(n)); return { sub: days === 1 ? 'Charts will show this 1 day.' : 'Charts will show these ' + days + ' days.', presets: [7, 30, 90].map((n) => { const on = to === T && from === T - n + 1; return { label: n + ' days', aria: 'Last ' + n + ' days', on, bg: on ? t.accent : 'transparent', fg: on ? t.onAccent : t.text2, fw: on ? '800' : '600', pick: () => this.setState({ popFrom: T - n + 1, popTo: T }) }; }), rows: [ { label: 'From', value: show(from), aria: 'From ' + k.full(from) + '. Tap to change' }, { label: 'To', value: show(to), aria: 'To ' + k.full(to) + '. Tap to change' } ] }; }
