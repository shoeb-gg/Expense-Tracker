// Month pop-up. The year button swaps the twelve months for twelve years in the very same tiles, so nothing else moves. popVals(t) { const k = this.popKit(t); const st = this.state || {}; const POP_VIEW = 'months'; const view = st.popView || POP_VIEW; const years = view === 'years'; const ym = st.popYM != null ? st.popYM : 2026 * 12 + 9; const selY = Math.floor(ym / 12); const selM = ym % 12; const y = st.popY != null ? st.popY : selY; const page = st.popPage != null ? st.popPage : 2018; const now = k.dt(k.TODAY); // The chosen tile is filled. The current month or year, when it is not the chosen one, is only a coloured label. const cell = (label, on, isNow, pick) => ({ label, on, bg: on ? t.accent : t.tile, fg: on ? t.onAccent : (isNow ? t.accent : t.text), fw: on || isNow ? '800' : '600', pick: this.popTap(pick) }); const prev = () => this.setState(years ? { popPage: page - 12 } : { popY: y - 1 }); const next = () => this.setState(years ? { popPage: page + 12 } : { popY: y + 1 }); // Swiping across the pop-up steps one year (twelve years in the year view), the same as the two arrows. const sw = this.popSwipe(prev, next); const cells = []; for (let i = 0; i < 12; i++) { if (years) { const yy = page + i; cells.push(cell(String(yy), yy === y, yy === now.getUTCFullYear(), () => this.setState({ popY: yy, popView: 'months' }))); } else { cells.push(cell(k.MS[i], y === selY && i === selM, y === now.getUTCFullYear() && i === now.getUTCMonth(), () => this.setState({ popYM: y * 12 + i }))); } } return { head: k.ML[selM] + ' ' + selY, year: years ? page + ' – ' + (page + 11) : String(y), caret: years ? 'expand_less' : 'expand_more', toggleLabel: years ? 'Back to months' : 'Choose year', isYears: years, gridLabel: years ? 'Years' : 'Months', cells, toggle: () => this.setState({ popView: years ? 'months' : 'years' }), prevLabel: years ? 'Earlier years' : 'Previous year', nextLabel: years ? 'Later years' : 'Next year', prev, next, swDown: sw.down, swUp: sw.up }; }
