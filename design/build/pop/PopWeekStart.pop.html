First day of the week Used by the calendar and the weekly chart.
// First day of the week. Same order as Money Tracker's sheet. A tap chooses and closes. popVals(t) { const k = this.popKit(t); const st = this.state || {}; const cur = st.popPick || 'Sunday'; return { grab: k.grab, opts: ['Monday', 'Saturday', 'Sunday'].map((label) => Object.assign(k.opt(label === cur), { label, pick: () => this.setState({ popPick: label }) })) }; }
