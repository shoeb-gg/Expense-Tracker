// Calculator keys: plus and minus only, or all four. popVals(t) { const k = this.popKit(t); const st = this.state || {}; const cur = st.popPick != null ? st.popPick : 0; // Both rows show all four keys in the same places; the ones a choice leaves out are faded. const KEYS = ['+', '\u2212', '\u00D7', '\u00F7']; const LIST = [['Plus and minus', 2], ['All four keys', 4]]; return { grab: k.grab, opts: LIST.map(([label, n], i) => Object.assign(k.opt(i === cur), { label, aria: label, keys: KEYS.map((ch, j) => ({ ch, op: j < n ? 1 : 0.3 })), pick: () => this.setState({ popPick: i }) })) }; }
